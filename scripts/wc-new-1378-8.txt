/** Standard defaults from client onboarding (Monthly / Quarterly) */
const ONBOARDING_GST_DEFAULTS = {
  Monthly: { gstr1Due: '11', gstr3bDue: '20', gstr9Due: '31' },
  Quarterly: { gstr1Due: '13', gstr3bDue: '22', gstr9Due: '31' },
} as const

function getGstRegistration(client: MockClient | undefined, gstin: string): GstRegistration | undefined {
  return client?.gstRegistrations.find(r => r.gstin === gstin)
}

/** Resolve filing due day from client onboarding config (fixed per company/GSTIN) */
function getGstDueDay(config: GstRegistration, returnType: string): number | null {
  if (returnType === 'GSTR-1') return parseInt(config.gstr1Due, 10) || null
  if (returnType === 'GSTR-3B') return parseInt(config.gstr3bDue, 10) || null
  if (returnType === 'GSTR-1A') {
    // GSTR-1A: 13th for monthly filers; quarterly uses same day as GSTR-1 (13th)
    return config.filingFrequency === 'Monthly' ? 13 : parseInt(config.gstr1Due, 10) || null
  }
  if (returnType === 'GSTR-9' || returnType === 'GSTR-9C') return parseInt(config.gstr9Due, 10) || 31
  return null
}

/** Compute full due date using fixed onboarding day-of-month + return period */
function gstFilingDueDate(
  config: GstRegistration | undefined,
  returnType: string,
  periodMonth: string,
): string {
  if (!config || !periodMonth || !returnType) return ''

  const parsed = new Date(`1 ${periodMonth}`)
  if (isNaN(parsed.getTime())) return ''
  const y = parsed.getFullYear()
  const m = parsed.getMonth()

  if (returnType === 'GSTR-9' || returnType === 'GSTR-9C') {
    const day = getGstDueDay(config, returnType) ?? 31
    return `${y + 1}-12-${String(day).padStart(2, '0')}`
  }

  const dueDay = getGstDueDay(config, returnType)
  if (!dueDay) return ''

  const nextM = (m + 1) % 12
  const nextY = m === 11 ? y + 1 : y
  return `${nextY}-${String(nextM + 1).padStart(2, '0')}-${String(dueDay).padStart(2, '0')}`
}

function formatDueDateLabel(iso: string): string {
  if (!iso) return '—'
  const d = new Date(iso + 'T00:00:00')
  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' })
}
