        else if (!/^[A-Z]{4}[0-9]{5}[A-Z]$/.test(form.tanNumber.toUpperCase())) errs.tanNumber = 'Invalid TAN format (AAAA99999A)'
      }
    }

        errs.modules = 'Enable at least one compliance module'
      }
